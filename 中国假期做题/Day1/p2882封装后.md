```C++
#include <iostream>
#include <cstdio>
#include <cstring>
using namespace std;

const int maxn = 10009;
int n;
bool A[maxn];   // 原始状态：0 = B, 1 = F
bool diff[maxn]; // 异或差分数组

// 判断固定长度 len 是否可行
// 可行返回翻转次数，不可行返回 -1
int solve(int len) {
    memset(diff, 0, sizeof diff);

    bool flip = 0; // 当前累计翻转状态
    int cnt = 0;

    for (int i = 1; i <= n; ++i) {
        flip ^= diff[i]; // 前缀异或，展开区间翻转状态

        if ((A[i] ^ flip) == 0) { // 实际方向是 B，需要翻转
            if (i + len - 1 > n) return -1; // 越界，无解

            flip ^= 1;          // 从 i 开始翻转
            diff[i + len] ^= 1; // 到 i+len 结束
            ++cnt;
        }
    }
    return cnt;
}

int main() {
    scanf("%d", &n);

    char ch;
    for (int i = 1; i <= n; ++i) {
        cin >> ch;
        A[i] = (ch == 'B' ? 0 : 1);
    }

    int mincnt = 0x7fffffff;
    int anslen = 1;

    for (int len = 1; len <= n; ++len) {
        int cnt = solve(len);
        if (cnt != -1 && cnt < mincnt) {
            mincnt = cnt;
            anslen = len;
        }
    }

    printf("%d %d\n", anslen, mincnt);
    return 0;
}#include <iostream>
```
