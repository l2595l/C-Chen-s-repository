```c++
#include <cstdio>
#include <algorithm>
using namespace std;

const int MAXN = 100005;      // n 最大 10^5，多开 5 个防越界
const int MOD  = 99999997;    // 题目要求的取模数

int n;                        // 火柴根数
int a[MAXN], b[MAXN];         // a[]: 第一列高度   b[]: 第二列高度
int id1[MAXN], id2[MAXN];     // 下标数组，排序后 id1[k] 表示第一列第 k 矮火柴的原位置
int x[MAXN];                  // 映射数组：把两列同序配对后转化成的排列
int bit[MAXN];                // 树状数组（Binary Indexed Tree）
long long ans = 0;            // 逆序对总数（可能超 int，用 long long）

/* ---------------- 快读 ---------------- */
// 用 getchar 逐字符读入，比 scanf 更快
inline int read() {
    int x = 0; char c = getchar();
    while (c < '0' || c > '9') c = getchar();       // 跳过非数字字符
    while (c >= '0' && c <= '9') {                  // 读入数字
        x = x * 10 + c - '0';
        c = getchar();
    }
    return x;
}

/* ---------------- 树状数组三件套 ---------------- */
// lowbit：取出 v 的二进制最低位的 1 所代表的值，如 lowbit(6)=2
inline int lowbit(int v) { return v & (-v); }

// 单点加 1：在位置 p 上加 1，并向上更新所有父节点
inline void add(int p) {
    for (; p <= n; p += lowbit(p)) ++bit[p];
}

// 前缀和：求 1..p 区间内已插入元素个数
inline int sum(int p) {
    int s = 0;
    for (; p > 0; p -= lowbit(p)) s += bit[p];
    return s;
}

/* ---------------- 主函数 ---------------- */
int main() {
    n = read();                                     // 读入 n

    // 读入第一列火柴高度，并初始化下标数组 id1[i] = i
    for (int i = 1; i <= n; ++i) { a[i] = read(); id1[i] = i; }

    // 读入第二列火柴高度，并初始化下标数组 id2[i] = i
    for (int i = 1; i <= n; ++i) { b[i] = read(); id2[i] = i; }

    // 对下标数组排序：按对应高度升序排
    // 排序后 id1[k] = 第一列第 k 矮的火柴的原位置
    //       id2[k] = 第二列第 k 矮的火柴的原位置
    sort(id1 + 1, id1 + n + 1, [](int i, int j){ return a[i] < a[j]; });
    sort(id2 + 1, id2 + n + 1, [](int i, int j){ return b[i] < b[j]; });

    // 核心转换：把"两列同序配对"转成排列问题
    // 含义：第二列原位置 id2[k] 处，最终应放第一列原位置 id1[k] 的火柴
    // 这样 x[] 就是一个 1..n 的排列，x[] 的逆序对数就是答案
    for (int k = 1; k <= n; ++k) x[id2[k]] = id1[k];

    // 从右往左扫描，用树状数组统计"右侧有多少比当前小的数"
    // 对位置 i：sum(x[i]-1) 就是已插入的、值 < x[i] 的元素个数
    // 即 (x[i], 后面的元素) 中，后面的更小的数 = 以 x[i] 为左端的逆序对数
    for (int i = n; i >= 1; --i) {
        ans += sum(x[i] - 1);   // 累加逆序对
        add(x[i]);              // 把当前值插入树状数组
    }

    printf("%lld\n", ans % MOD);   // 按题目要求取模输出
    return 0;
}
```

│
▼
构造下标数组 id1[], id2[]
│
▼
按高度升序排序下标（间接排序）
│
▼
构造排列 x[id2[k]] = id1[k]
│
▼
从右往左扫描 x[]，树状数组累加逆序对
│
▼
输出 ans % MOD
