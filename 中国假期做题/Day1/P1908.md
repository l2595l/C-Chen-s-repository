输出格式
输出序列中逆序对的数目。
```c++
#include <cstdio>
#include <iostream>
using namespace std;

#define ll long long  // 宏定义，后续用 ll 代替 long long，方便书写

// 题目中 n <= 500000，所以数组必须开到 50万+10，否则会 RE（运行时错误/越界）
// 全局变量在堆区，开大数组不会导致栈溢出（不要开在 main 函数内部）
const int MAXN = 500010; 
int n, a[MAXN], c[MAXN];  // a是原数组，c是归并排序用的临时数组
ll ans = 0;               // 逆序对总数，必须用 long long，因为最大可达到 n*(n-1)/2 ≈ 1.25*10^11，int 会溢出

// 归并排序函数：对数组 a 的 [b, e] 区间进行升序排序，并统计逆序对
void msort(int b, int e) {
    // 递归终止条件：区间内只有一个元素或没有元素，天然有序
    if (b >= e) return;

    // 1. 分治：找到中点，将区间一分为二
    int mid = (b + e) / 2;
    msort(b, mid);      // 递归处理左半区间 [b, mid]
    msort(mid + 1, e);  // 递归处理右半区间 [mid+1, e]

    // 2. 合并：将两个有序子数组合并为一个有序数组
    int i = b;          // i 指向左半区间的起始位置
    int j = mid + 1;    // j 指向右半区间的起始位置
    int k = b;          // k 指向临时数组 c 的起始位置

    // 双指针合并：只要左右两边都有元素没处理完
    while (i <= mid && j <= e) {
        // 如果左半部分的元素小于等于右半部分的元素，不构成逆序对
        if (a[i] <= a[j]) {
            c[k++] = a[i++];  // 将较小的 a[i] 放入临时数组，i 和 k 后移
        } 
        // 如果左半部分大于右半部分，构成了逆序对
        else {
            c[k++] = a[j++];  // 先把较小的 a[j] 放入临时数组，j 和 k 后移
            
            // 【核心逻辑】既然 a[i] > a[j]，又因为左半区间 [i, mid] 已经是升序的
            // 所以左半区间从 i 到 mid 的所有元素都严格大于 a[j]
            // 这部分元素的个数就是：mid - i + 1
            ans += mid - i + 1; 
        }
    }

    // 3. 收尾：将左半区间或右半区间剩余的元素直接追加到临时数组
    while (i <= mid) c[k++] = a[i++];  // 左半部分还有剩余
    while (j <= e) c[k++] = a[j++];    // 右半部分还有剩余

    // 4. 将排好序的临时数组 c 的内容，拷贝回原数组 a 的 [b, e] 区间
    // 这样上一层递归才能继续用排好序的子数组进行合并
    for (int t = b; t <= e; ++t) {
        a[t] = c[t];
    }
}

int main() {
    // 读入数据个数
    scanf("%d", &n);
    
    // 读入具体数据，注意下标从 1 开始（方便与归并排序的区间参数对应）
    for (int i = 1; i <= n; ++i) {
        scanf("%d", &a[i]);
    }

    ans = 0;           // 初始化逆序对计数
    msort(1, n);       // 对 [1, n] 区间进行归并排序，排序过程中统计逆序对

    // 输出结果，long long 对应的格式说明符是 %lld
    printf("%lld\n", ans);

    return 0;
}
```
