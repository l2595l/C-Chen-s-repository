/**
 * Johnson算法完整流程
 *
 * 输入：图G(V,E)，可能有负权边
 * 输出：所有顶点对的最短距离
 */
```C++
class Johnson {
private:
    int n;  // 顶点数
    vector<vector<pair<int, int>>> adj;  // 原图
    vector<long long> h;  // 势能
    vector<vector<long long>> dist;  // 结果

public:
    Johnson(int n) : n(n) {
        adj.resize(n);
        dist.resize(n, vector<long long>(n, INF));
    }

    void addEdge(int u, int v, int w) {
        adj[u].push_back({v, w});
    }

    /**
     * 步骤1：添加超级源点，运行Bellman-Ford
     * 计算势能 h[v]
     */
    bool bellmanFordForPotential() {
        // 创建超级源点s，连接到所有顶点，边权为0
        vector<long long> dis(n + 1, INF);
        dis[n] = 0;  // 超级源点编号为n

        // 松弛 n 次
        for (int i = 0; i <= n; i++) {
            bool updated = false;

            // 从超级源点出发的边（权值为0）
            for (int v = 0; v < n; v++) {
                if (dis[n] + 0 < dis[v]) {
                    dis[v] = dis[n] + 0;
                    updated = true;
                }
            }

            // 原图的边
            for (int u = 0; u < n; u++) {
                if (dis[u] == INF) continue;
                for (auto [v, w] : adj[u]) {
                    if (dis[v] > dis[u] + w) {
                        dis[v] = dis[u] + w;
                        updated = true;
                    }
                }
            }

            // 第n轮还能更新 → 存在负环
            if (i == n && updated) {
                return false;
            }

            if (!updated) break;
        }

        // 存储势能（去掉超级源点）
        h.resize(n);
        for (int i = 0; i < n; i++) {
            h[i] = dis[i];
        }
        return true;
    }

    /**
     * 步骤2：重新赋予权重
     */
    vector<vector<pair<int, int>>> reweight() {
        vector<vector<pair<int, int>>> newAdj(n);

        for (int u = 0; u < n; u++) {
            for (auto [v, w] : adj[u]) {
                // 新边权 = 原边权 + h[u] - h[v]
                long long newW = w + h[u] - h[v];
                // 应该 >= 0
                newAdj[u].push_back({v, (int)newW});
            }
        }

        return newAdj;
    }

    /**
     * 步骤3：对每个顶点运行Dijkstra
     */
    void dijkstra(int s, const vector<vector<pair<int, int>>>& newAdj) {
        vector<long long> d(n, INF);
        priority_queue<pair<long long, int>, vector<pair<long long, int>>, greater<>> pq;

        d[s] = 0;
        pq.push({0, s});

        while (!pq.empty()) {
            auto [dist_u, u] = pq.top();
            pq.pop();

            if (dist_u > d[u]) continue;

            for (auto [v, w] : newAdj[u]) {
                if (d[v] > d[u] + w) {
                    d[v] = d[u] + w;
                    pq.push({d[v], v});
                }
            }
        }

        // 步骤4：还原真实距离
        for (int v = 0; v < n; v++) {
            if (d[v] != INF) {
                dist[s][v] = d[v] - h[s] + h[v];
            }
        }
    }

    /**
     * 运行Johnson算法
     */
    bool johnson() {
        // 步骤1：计算势能
        if (!bellmanFordForPotential()) {
            cout << "图中存在负环！" << endl;
            return false;
        }

        // 步骤2：重新赋予权重
        auto newAdj = reweight();

        // 步骤3：对每个顶点运行Dijkstra
        for (int s = 0; s < n; s++) {
            dijkstra(s, newAdj);
        }

        return true;
    }

    void printDistances() {
        cout << "所有顶点对最短距离：" << endl;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (dist[i][j] == INF) {
                    cout << "∞ ";
                } else {
                    cout << dist[i][j] << " ";
                }
            }
            cout << endl;
        }
    }
};
```
